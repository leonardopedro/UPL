\usepackage[utf8]{inputenc}

% Set & as a 'letter' (catcode 11) to prevent it from being treated as 
% an alignment tab inside smodule and other environments.
\catcode`&=11

% Helper to set UPL-specific catcodes (useful inside environments)
\newcommand{\uplsetup}{%
  \catcode`&=11
  \catcode`_=12
  \catcode`^=12
}

% UPL Language definition for listings
\lstdefinelanguage{UPL}{
  keywords={theory, type, include, forall, exists, module, |-},
  sensitive=true,
  comment=[l]{//},
  morecomment=[s]{/*}{*/},
  morestring=[b]"
}

\makeatletter
% upl environment for code sections
\lstnewenvironment{upl}[1][]
  {\lstset{
    language=UPL,
    basicstyle=\small\ttfamily,
    frame=single,
    breaklines=true,
    literate={=>}{{$\Rightarrow$}}2,
    #1
  }}
  {}
\makeatother

% uplinline command for inline code
\newcommand{\uplinline}{\lstinline[language=UPL,basicstyle=\ttfamily]}
